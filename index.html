// ACTUALIZACIÓN DE LA GRÁFICA MOLLIER function actualizarTodo() { const refKey = document.getElementById('refSelect').value; const refData = refrigerantes[refKey]; const pf = pUnits[pUnitActual].factor; // Lectura Inputs Presión -> Convertir a bar relativo let pSuctInput = parseFloat(document.getElementById('pSuct').value); let pDescInput = parseFloat(document.getElementById('pDesc').value); if (isNaN(pSuctInput)) pSuctInput = 3.5 * pf; if (isNaN(pDescInput)) pDescInput = 18.0 * pf; let pSuctBarRel = pSuctInput / pf; let pDescBarRel = pDescInput / pf; if (pSuctBarRel < 0.1) pSuctBarRel = 0.1; if (pDescBarRel <= pSuctBarRel) pDescBarRel = pSuctBarRel + 2.0; // Presión Absoluta para el Diagrama de Mollier const pSuctBarAbs = pSuctBarRel + 1.013; const pDescBarAbs = Math.min(pDescBarRel + 1.013, refData.criticalP - 0.2); // Lectura Inputs Temperatura let tSuctVal = parseFloat(document.getElementById('tSuct').value) || 10; let tLiqVal = parseFloat(document.getElementById('tLiq').value) || 35; let tDescRealVal = parseFloat(document.getElementById('tDescReal').value) || 75; if (tUnitActual === 'F') { tSuctVal = (tSuctVal - 32) * 5/9; tLiqVal = (tLiqVal - 32) * 5/9; tDescRealVal = (tDescRealVal - 32) * 5/9; } // Lectura Mediciones Eléctricas const vVolt = parseFloat(document.getElementById('vVolt').value) || 220; const iAmp = parseFloat(document.getElementById('iAmp').value) || 0; const pKw = (vVolt * iAmp * 0.9) / 1000.0; // Propiedades de Saturación const propEvap = getPropAtP(refData, pSuctBarAbs); const propCond = getPropAtP(refData, pDescBarAbs); // Cálculos Recalentamiento y Subenfriamiento const superHeatK = Math.max(tSuctVal - propEvap.tSat, 0); const subCoolK = Math.max(propCond.tSat - tLiqVal, 0); // Entalpías del ciclo (kJ/kg) const h1 = propEvap.hV + (superHeatK * 1.25); const h2 = propCond.hV + Math.max(tDescRealVal - propCond.tSat, 5) * 1.35; const h3 = propCond.hL - (subCoolK * 1.25); const h4 = h3; // KPIs const qn = h1 - h4; const w = h2 - h1; const cop = w > 0 ? (qn / w) : 0; document.getElementById('kpiSH').innerText = `${superHeatK.toFixed(1)} K`; document.getElementById('kpiSC').innerText = `${subCoolK.toFixed(1)} K`; document.getElementById('kpiKW').innerText = `${pKw.toFixed(2)} kW`; document.getElementById('kpiqn').innerText = `${qn.toFixed(1)} kJ/kg`; document.getElementById('kpiw').innerText = `${w.toFixed(1)} kJ/kg`; document.getElementById('kpicop').innerText = cop.toFixed(2); // Trazos en la unidad seleccionada const pSuctPlot = pSuctBarAbs * pf; const pDescPlot = pDescBarAbs * pf; // Trazado del Ciclo Operativo Real const cycleTrace = { x: [h1, h2, h3, h4, h1], y: [pSuctPlot, pDescPlot, pDescPlot, pSuctPlot, pSuctPlot], mode: 'lines+markers+text', name: 'Ciclo Operativo Real', text: ['1 (Succ)', '2 (Desc)', '3 (Liq)', '4 (Evap)', ''], textposition: ['bottom right', 'top right', 'top left', 'bottom left', ''], textfont: { color: '#f8fafc', size: 11 }, line: { color: '#ef4444', width: 2.5 }, marker: { size: 6, color: '#f59e0b' } }; const traces = [getDomeTrace(refData), cycleTrace]; // Sombra interna del área de trabajo traces.unshift({ x: [h1, h2, h3, h4], y: [pSuctPlot, pDescPlot, pDescPlot, pSuctPlot], fill: 'toself', fillcolor: 'rgba(56, 189, 248, 0.15)', line: { color: 'transparent' }, name: 'Zona Operativa', showlegend: false }); // DEFINICIÓN DE LIMITES FIJOS (DOMO COMPLETO vs ZONA ACTIVA) let xRangeFinal, yRangeFinal; const maxH_Ciclo = Math.max(h1, h2, h3, h4); const minH_Ciclo = Math.min(h1, h2, h3, h4); if (zonaActivaHabilitada) { // MODO ZONA ACTIVA: Zoom enfocado solo en el ciclo const hMargin = (maxH_Ciclo - minH_Ciclo) * 0.25; const minPLog = Math.log10(Math.min(pSuctPlot, pDescPlot)) - 0.15; const maxPLog = Math.log10(Math.max(pSuctPlot, pDescPlot)) + 0.15; xRangeFinal = [minH_Ciclo - hMargin, maxH_Ciclo + hMargin]; yRangeFinal = [minPLog, maxPLog]; } else { // MODO NORMAL (OFF): Vista completa del Domo con el ciclo visible const xLimitMax = Math.max(refData.xRange[1], maxH_Ciclo + 30); xRangeFinal = [refData.xRange[0], xLimitMax]; yRangeFinal = [ Math.log10(refData.pRange[0] * pf), Math.log10(refData.pRange[1] * pf) ]; } // CONFIGURACIÓN DEL LAYOUT (TOTALMENTE ESTÁTICO) const layout = { margin: { l: 50, r: 20, t: 30, b: 40 }, autosize: true, paper_bgcolor: '#1e293b', plot_bgcolor: '#0f172a', dragmode: false, // <-- DESACTIVA EL ARRASTRE / MOVIMIENTO CON EL MOUSE O TÁCTIL title: { text: `Diagrama Mollier p-h (${refData.nombre}) - Operación Real`, font: { color: '#f8fafc', size: 13 } }, xaxis: { title: { text: 'Entalpía h [kJ/kg]', font: { color: '#94a3b8', size: 11 } }, range: xRangeFinal, fixedrange: true, // <-- BLOQUEA EL EJE X CONTRA ZOOM MANUAL gridcolor: '#334155', tickfont: { color: '#94a3b8' } }, yaxis: { title: { text: `Presión P_abs [${pUnits[pUnitActual].label}]`, font: { color: '#94a3b8', size: 11 } }, type: 'log', range: yRangeFinal, fixedrange: true, // <-- BLOQUEA EL EJE Y CONTRA ZOOM MANUAL gridcolor: '#334155', tickfont: { color: '#94a3b8' } }, legend: { x: 0.02, y: 0.98, font: { color: '#f8fafc', size: 10 }, bgcolor: 'rgba(15, 23, 42, 0.6)' } }; // OPCIONES DE PLOTLY (Sin botones ni interacción) const config = { responsive: true, displayModeBar: false, // Oculta la barra flotante de herramientas scrollZoom: false // Desactiva el zoom con la rueda del mouse o gestos táctiles }; Plotly.react('mollierPlot', traces, layout, config); }
